  <script>
    // Toolbox: the menu of blocks
    const toolboxJson = {
      "kind": "categoryToolbox",
      "contents": [
        {
          "kind": "category", "name": "Robot", "colour": "20",
          "contents": [
            { "kind": "block", "type": "robot_forward" },
            { "kind": "block", "type": "robot_turn" }
          ]
        },
        {
          "kind": "category", "name": "Loops", "colour": "120",
          "contents": [
            {
              "kind": "block", "type": "controls_repeat_ext",
              "inputs": { "TIMES": { "shadow": { "type": "math_number", "fields": { "NUM": 4 } } } }
            }
          ]
        },
        {
          "kind": "category", "name": "My Custom Blocks", "colour": "120",
          "contents": [
            { "kind": "block", "type": "my_simple_block" }
          ]
        }
      ]
    };

    const workspace = Blockly.inject('blocklyDiv', {
      toolbox: toolboxJson,
      trashcan: true,
      media: 'blockly/media/'   // sounds and icons come from this folder, not the internet
    });

    let code = '';

    // Update the code panel whenever the blocks change
    workspace.addChangeListener((e) => {
      if (e.isUiEvent) return;
      code = Blockly.JavaScript.workspaceToCode(workspace);
      document.getElementById('codeArea').textContent = code;
    });

    // Run the generated code. forward() and turn() are stand-ins that print
    // to the output panel; swap them for real robot commands later.
    document.getElementById('runButton').addEventListener('click', () => {
      const out = document.getElementById('output');
      out.textContent = '';
      const say = (...args) => { out.textContent += args.join(' ') + '\n'; };
      try {
        new Function('console', 'forward', 'turn', code)(
          { log: say },
          (cm) => say('forward ' + cm + ' cm'),
          (deg) => say('turn ' + deg + ' degrees')
        );
      } catch (err) {
        say('Error: ' + err.message);
      }
    });
  </script>
